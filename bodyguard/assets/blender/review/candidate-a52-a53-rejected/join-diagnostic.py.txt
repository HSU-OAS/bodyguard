import json,bpy,bmesh,math; from statistics import mean
report=json.load(open("/private/tmp/bodyguard-tailor.6b7Uvu/a52-generation.json"))
for info in report:
 obj=bpy.data.objects["BODYGUARD "+info["body"]+" sculpt WIP"]
 bm=bmesh.new();bm.from_mesh(obj.data);bm.normal_update()
 rows={}
 for key,z in (("body",info["cut_m"]),("head",info["head_cut_m"])):
  edges=[e for e in bm.edges if abs(e.verts[0].co.z-z)<1e-6 and abs(e.verts[1].co.z-z)<1e-6 and len(e.link_faces)==2]
  angles=sorted(math.degrees(e.calc_face_angle()) for e in edges)
  rows[key]={"count":len(angles),"mean_deg":mean(angles),"p95_deg":angles[int(.95*(len(angles)-1))],"max_deg":max(angles),"folds_over90_long_1mm":sum(e.calc_length()>.001 and e.calc_face_angle()>math.pi/2 for e in edges),"max_edge_m":max(e.calc_length() for e in edges),"min_face_area_m2":min(f.calc_area() for e in edges for f in e.link_faces)}
 print("NATIVE JOIN TANGENT DIAGNOSTIC",info["body"],json.dumps(rows))
 bm.free()
print("NATIVE SMOOTH",bmesh.ops.smooth_vert.__doc__)

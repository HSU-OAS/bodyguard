"""FR-A-03: actual native cut regression; tiny surfaces are not body generators."""

import importlib.util
import json
import sys
from pathlib import Path

import bmesh
from mathutils import Vector


def module(name, filename):
    spec = importlib.util.spec_from_file_location(
        name, Path(__file__).with_name(filename)
    )
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value


current = module("oblique_candidate", "a72c-sculpted-section-bridge-fairing.py")
prior = module("horizontal_candidate", "a64-continuous-jaw-neutral-head.py")
helper = module("unchanged_boundary", "a49-density-producer.py")
square = [(-2, -2), (2, -2), (2, 2), (-2, 2)]
nonstar = [
    (-2, -2),
    (2, -2),
    (2, 2),
    (0.5, 2),
    (0.5, -0.5),
    (-0.5, -0.5),
    (-0.5, 2),
    (-2, 2),
]
output = Path(sys.argv[sys.argv.index("--") + 1])
assert output.suffix == ".json" and not output.exists()
checks = []


def fixture(contours, planes, slopes):
    mesh = bmesh.new()
    original = mesh.verts.layers.int.new("Fixture original")
    deform = mesh.verts.layers.deform.new()
    edges = []
    for points, z, slope in zip(contours, planes, slopes, strict=True):
        top = [
            mesh.verts.new((x * 0.01, y * 0.01, z + slope * y * 0.01))
            for x, y in points
        ]
        bottom = [mesh.verts.new((v.co.x, v.co.y, v.co.z - 0.01)) for v in top]
        for i, vertex in enumerate(top):
            following = (i + 1) % len(top)
            mesh.faces.new((vertex, top[following], bottom[following], bottom[i]))
            edges.append(mesh.edges.get((vertex, top[following])))
    mesh.verts.index_update()
    for vertex in mesh.verts:
        vertex[original] = vertex.index + 1
        vertex[deform][0] = 1
    return mesh, edges, original


for name, contours, planes, slopes, expected_error in (
    ("nonstar-horizontal", [nonstar, square], [1, 1.04], [0, 0], False),
    (
        "reverse-order-oblique",
        [list(reversed(nonstar)), square],
        [1, 1.04],
        [0.22, 0],
        False,
    ),
    ("different-counts", [square, nonstar], [1, 1.04], [0, 0], False),
    ("parallel-oblique", [square, square], [1, 1.04], [0.22, 0.22], False),
    ("steeper-parallel-oblique", [square, nonstar], [1, 1.04], [0.6, 0.6], False),
    ("same-plane-rejected", [square, square], [1, 1], [0, 0], True),
    ("third-ring-rejected", [square, square, square], [1, 1.04, 1.08], [0, 0, 0], True),
    ("crossing-planes-rejected", [square, square], [1, 1.001], [0.22, 0], True),
):
    mesh, edges, original = fixture(contours, planes, slopes)
    rejected = False
    try:
        result = current.align_neck_boundary_samples(
            mesh, edges, original, helper.closed_loops
        )
        assert result["after"][0] == result["after"][1]
        assert all(
            abs(sum(v[mesh.verts.layers.deform.active].values()) - 1) < 1e-6
            for v in mesh.verts
        )
    except AssertionError:
        rejected = True
    assert rejected == expected_error, name
    checks.append({"name": name, "passed": True, "expected_rejection": rejected})
    mesh.free()

mesh, edges, original = fixture([square, square], [1, 1.04], [0, 0])
edges[0].verts[0].co.z += 0.002
try:
    current.align_neck_boundary_samples(mesh, edges, original, helper.closed_loops)
except AssertionError:
    checks.append(
        {"name": "nonplanar-rejected", "passed": True, "expected_rejection": True}
    )
else:
    raise AssertionError("Reject a nonplanar native contour")
mesh.free()

mesh, edges, original = fixture([nonstar, square], [1, 1.04], [0.22, 0])
try:
    prior.align_neck_boundary_samples(mesh, edges, original, helper.closed_loops)
except AssertionError:
    checks.append({"name": "legacy-oblique-failure-reproduced", "passed": True})
else:
    raise AssertionError("Prior constant-Z method must reject the actual oblique input")
mesh.free()
# FR-A-03: tiny native regression surfaces are not generated avatar bodies.
for slope, dy in ((0.6, -0.03), (0.22, -0.02)):
    normal, intercept = Vector((0, -slope, 1)), 0.02
    mesh = bmesh.new()
    bmesh.ops.create_cube(mesh, size=0.2)
    deform = mesh.verts.layers.deform.new()
    for v in mesh.verts:
        v[deform][0] = 1
        v.co.y += dy
    result = current.bisect_registered_head(
        mesh, normal, intercept, helper.closed_loops
    )
    assert result["final_contour_points"] >= 4
    assert abs(result["final_plane_intercept_m"] - intercept) < 1e-6
    checks.append({"name": f"registered-final-plane-{slope}", "passed": True})
    mesh.free()

mesh = bmesh.new()
bmesh.ops.create_cube(mesh, size=0.2)
deform = mesh.verts.layers.deform.new()
for v in mesh.verts:
    v[deform][0] = 1
normal, intercept = Vector((0, -0.6, 1)), 0.02
current.bisect_registered_head(mesh, normal, intercept, helper.closed_loops)
for v in mesh.verts:
    v.co.y -= 0.03
ring = helper.closed_loops([e for e in mesh.edges if e.is_boundary])[0]
assert all(abs(normal.dot(v.co) - intercept - 0.018) < 1e-6 for v in ring)
checks.append(
    {
        "name": "early-cut-translation-plane-drift-reproduced",
        "passed": True,
        "artist_plane_drift_m": 0.018,
    }
)
mesh.free()
# FR-A-03: native regression points, not a body or an anatomical target.
for slope, reference, target in ((0.6, 1.54, 1.517), (0.22, 1.50, 1.517)):
    normal = Vector((0, -slope, 1))
    xy = (0.003, -0.02)
    translation = current.register_artist_section(normal, xy, reference, target)
    points = [Vector((x, y, reference + slope * y)) for x, y in ((0, 0), (0.01, -0.02))]
    assert all(abs(normal.dot(point + translation) - target) < 1e-6 for point in points)
    checks.append({"name": f"artist-section-registration-{slope}", "passed": True})
try:
    current.register_artist_section(Vector((0, 0, 0)), (0, 0), 1.54, 1.517)
except AssertionError:
    checks.append({"name": "invalid-artist-plane-rejected", "passed": True})
else:
    raise AssertionError("Do not register a degenerate artist plane")
output.write_text(
    json.dumps(
        {"passed": True, "artist_fixture_only": True, "checks": checks}, indent=2
    )
    + "\n"
)
print("ACTUAL NATIVE OBLIQUE REGRESSION", json.dumps(checks))

  for (const [i, edges] of [[0, 441], [1, 443]]) ok(d.rows[i].rings.every(r => r.edges === edges) && Math.abs(d.rows[i].actual_artist_vertical_gap_m - 0.024) < 1e-7, 'actual fixed24mm native ring');

for (const filename of ['draft-a72-lower-head-cage.blend', 'a72-generation-report.json', 'a72-diagnostic-unsaved.blend', 'a72-diagnostic-unsaved-report.json']) ok(!fs.existsSync(path.join(tmp, filename)), 'failed native output absent');
ok(json(folder + 'a72/native-boundary-check.json').checks.length === 13, 'legacy13 standalone fixtures not source approval');
const negative = json(folder + 'initial-inspector-ruff-negative.json');
ok(negative.length === 1 && negative[0].code === 'I001', 'actual original import issue retained');
for (const n of [0, 6, 12, '12-verified']) {
  const d = json(folder + 'inspection-' + n + '/inspection-report.json');
  ok(!d.source_saved && d.artist_inspection_only && !d.rig_deformation_or_release_approval && d.vertices === 67586 && d.head_cage_sculpt.upper_cage_and_topology_preserved, 'head inspection only');

}
const preservation = json(folder + 'a72c/authored-body-preservation.json');
ok(preservation.passed && !preservation.source_saved && preservation.rows[0].retained_authored_body_points_and_named_skin === 75915 && preservation.rows[1].retained_authored_body_points_and_named_skin === 75865 && preservation.rows.every(r => r.hands_163bones_rest_matrices_identical), 'independent actual retained body/named skin/hands/rig');

const files = [], markdown = ['bodyguard/docs/avatar-production.md', 'bodyguard/docs/stories/stage2-avatar-candidate.md'];
function walk(relative) {
  for (const e of fs.readdirSync(path.join(root, relative), { withFileTypes: true })) {
    if (e.name === 'evidence-audit.json') continue;
    const p = relative + e.name;
    if (e.isDirectory()) { walk(p + '/'); continue; }
    ok(e.isFile(), 'regular immutable evidence file');
    const b = read(p);
    files.push({ file: p, bytes: b.length, sha256: sha(b) });
    if (e.name.endsWith('.md')) markdown.push(p);
    if (!e.name.endsWith('.png')) continue;
    ok(b.subarray(0, 8).equals(Buffer.from([137,80,78,71,13,10,26,10])), 'actual PNG bytes');
    let native;
    const rel = p.slice(folder.length);
    if (rel.startsWith('inspection-')) native = path.join(tmp, 'a72-head-cage-' + rel);
    else if (rel.startsWith('a72b/first/')) native = path.join(tmp, 'first-a72b-sculpted-section-head', e.name);
    else if (rel.startsWith('a72c/source/')) native = path.join(tmp, 'source-a72c-full-review', e.name);
    else if (rel.startsWith('a72c/browser/')) native = path.join(root, 'bodyguard/frontend/test-results/avatar/avatar-A-06-실제-남녀-GLB-두-복제본을-로드하고-한쪽-기술-변형이-다른-뼈대를-바꾸지-않는다', (e.name.startsWith('male') ? '남성' : '여성') + '-two-models.png');
    else throw Error('Unexpected PNG origin');
    ok(b.equals(fs.readFileSync(native)), 'exact actual render/screenshot copy');
    png++;
  }
}

  const resolved = path.resolve(root, path.dirname(file), target);
  if (resolved === path.join(root, folder, 'evidence-audit.json')) continue;
  ok(fs.existsSync(resolved), 'local evidence link'); links++;
}

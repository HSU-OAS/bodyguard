"""FR-A-02/03: read-only retained body cross-sections; not clinical targets."""

import importlib.util
import json
import sys
from hashlib import file_digest
from pathlib import Path

import bmesh
import bpy

generation, output = map(Path, sys.argv[sys.argv.index("--") + 1 :])
if output.exists():
    raise FileExistsError("Preserve prior diagnostic evidence")
helper_path = Path(__file__).with_name("a49-density-producer.py")
with helper_path.open("rb") as file:
    assert file_digest(file, "sha256").hexdigest() == (
        "6a96114e186b0db44a622beaef770dc58a193656344586f9f189783cc14018c4"
    )
spec = importlib.util.spec_from_file_location("a49_native_boundary", helper_path)
helper = importlib.util.module_from_spec(spec)
spec.loader.exec_module(helper)
reports = []
for info in json.loads(generation.read_text()):
    body = bpy.data.objects[f"BODYGUARD {info['body']} sculpt WIP"]
    mesh = bmesh.new()
    mesh.from_mesh(body.data)
    sections = []
    for offset in (-0.030, -0.020, -0.010, 0.0):
        plane = info["cut_m"] + offset
        part = mesh.copy()
        bmesh.ops.bisect_plane(
            part,
            geom=list(part.verts) + list(part.edges) + list(part.faces),
            plane_co=(0, 0, plane),
            plane_no=(0, 0, 1),
            dist=1e-8,
            clear_outer=True,
        )
        edges = [e for e in part.edges if e.is_boundary]
        loops = helper.closed_loops(edges)
        assert len(loops) == 1, "Only one isolated native neck contour is supported"
        ring = loops[0]
        assert all(abs(v.co.z - plane) < 1e-6 for v in ring)
        ranges = [[min(v.co[a] for v in ring), max(v.co[a] for v in ring)] for a in (0, 1)]
        sections.append(
            {
                "plane_m": plane,
                "offset_from_current_cut_m": offset,
                "contour_vertices": len(ring),
                "xy_range_m": ranges,
                "width_depth_m": [high - low for low, high in ranges],
            }
        )
        part.free()
    mesh.free()
    reports.append({"body": info["body"], "sections": sections})
output.write_text(
    json.dumps({"source_saved": False, "artist_model_only": True, "reports": reports}, indent=2)
    + "\n"
)
print("ACTUAL RETAINED BODY SECTION PROBE", json.dumps(reports))

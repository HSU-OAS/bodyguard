"""FR-A-03: read-only source comparison; artist bridge, not motion approval."""

import json
import sys
from collections import Counter
from hashlib import file_digest
from pathlib import Path

import bpy
from mathutils import Vector

linear, surface, generation, output = map(Path, sys.argv[sys.argv.index("--") + 1 :])
assert output.suffix == ".json" and not output.exists()
expected = (
    "d0c55dc6f25a034a9f1fac59e26022841e5598c2948e68947b46df60deb39058",
    "6383829d3b631bf22c66bbce5472806f7fe9525dd3b6c554ad96ef4be3b5e0a5",
)
reports = json.loads(generation.read_text())
snapshots = []
for source, digest in zip((linear, surface), expected, strict=True):
    with source.open("rb") as file:
        assert file_digest(file, "sha256").hexdigest() == digest
    bpy.ops.wm.open_mainfile(filepath=str(source))
    result = []
    for report in reports:
        label = report["body"]
        obj = bpy.data.objects[f"BODYGUARD {label} sculpt WIP"]
        rig = bpy.data.objects[f"BODYGUARD {label} rig WIP"]
        normal = Vector(report["jaw_cut_plane_normal"])
        points = [tuple(v.co) for v in obj.data.vertices]
        skin = [
            tuple(sorted((obj.vertex_groups[g.group].name, g.weight) for g in v.groups))
            for v in obj.data.vertices
        ]
        fixed = Counter(
            (point, weights)
            for point, weights in zip(points, skin, strict=True)
            if normal.dot(Vector(point)) <= report["cut_m"] + 1e-6
            or normal.dot(Vector(point)) >= report["head_cut_m"] - 1e-6
        )
        hands = [(points[i], skin[i]) for i in obj["bodyguard_distal_hand_vertices"]]
        bones = [
            (b.name, tuple(tuple(row) for row in b.matrix_local))
            for b in rig.data.bones
        ]
        result.append(
            {
                "points": points,
                "skin": skin,
                "fixed": fixed,
                "hands": hands,
                "bones": bones,
                "body_matrix": tuple(tuple(row) for row in obj.matrix_world),
                "rig_matrix": tuple(tuple(row) for row in rig.matrix_world),
            }
        )
    snapshots.append(result)
rows = []
for label, before, after in zip(("Male", "Female"), *snapshots, strict=True):
    assert before["fixed"] == after["fixed"]
    for key in ("skin", "hands", "bones", "body_matrix", "rig_matrix"):
        assert before[key] == after[key], key
    assert len(before["points"]) == len(after["points"])
    deltas = [
        (Vector(b) - Vector(a)).length
        for a, b in zip(before["points"], after["points"], strict=True)
    ]
    assert max(deltas) > 1e-5
    rows.append(
        {
            "body": label,
            "vertices": len(deltas),
            "fixed_artist_points_and_skin": sum(before["fixed"].values()),
            "changed_bridge_vertices": sum(d > 1e-6 for d in deltas),
            "maximum_bridge_displacement_m": max(deltas),
            "skin_hands_163bones_rest_matrices_identical": True,
        }
    )
for source, digest in zip((linear, surface), expected, strict=True):
    with source.open("rb") as file:
        assert file_digest(file, "sha256").hexdigest() == digest
output.write_text(
    json.dumps(
        {
            "passed": True,
            "source_saved": False,
            "source_sha256": expected,
            "artist_geometry_only": True,
            "rows": rows,
        },
        indent=2,
    )
    + "\n"
)
print("ACTUAL READ-ONLY NATIVE BRIDGE COMPARISON", rows)

// FR-M-02/07: actual React/fake-device SDK retry after controlled pre-SDK loading failures.
import assert from 'node:assert/strict'
import { execFileSync } from 'node:child_process'
import { createHash } from 'node:crypto'
import { existsSync, readFileSync, writeFileSync } from 'node:fs'
import { createServer as createHTTPServer } from 'node:http'
import { resolve } from 'node:path'
import { chromium } from '@playwright/test'
import { createServer, mergeConfig } from 'vite'
import config from '../vite.config.ts'
import { SDK_DEV_POLICY } from './sdk-network-policy.ts'

const [scratch, label, variant] = process.argv.slice(2)
assert(/^\/private\/tmp\/bodyguard-worker-boot\.[A-Za-z0-9]+$/.test(scratch))
assert(/^[a-z][a-z0-9-]+$/.test(label) && !existsSync(`${scratch}/${label}.json`))
assert(variant === undefined || variant === 'baseline')
const original = variant
  ? execFileSync('git', ['show', '203a6ab:bodyguard/frontend/src/pose/workerInference.ts'], {
      encoding: 'utf8',
    })
  : null
const sha = (b) => createHash('sha256').update(b).digest('hex')
const sources = [
  'src/pose/workerInference.ts',
  'src/pose/nativeInference.worker.ts',
  'src/pose/workerInferenceRuntime.ts',
  'src/pose/workerNativeEngine.ts',
  'scripts/test-worker-boot-recovery.mjs',
  'pnpm-lock.yaml',
]
const snapshot = () => Object.fromEntries(sources.map((file) => [file, sha(readFileSync(file))]))
const before = snapshot(),
  cases = [],
  traces = []
let mode = 'normal',
  browser,
  failure = null,
  visibleError = null,
  external = 0
const errors = []
const api = createHTTPServer((req, res) => {
  res.setHeader('Content-Type', 'application/json')
  res.setHeader('Cache-Control', 'no-store')
  if (req.url === '/api/v1/me') res.end('{"id":"boot-fixture","email":"fixture@example.com"}')
  else {
    res.statusCode = 404
    res.end('{}')
  }
})
await new Promise((yes, no) => {
  api.once('error', no)
  api.listen(0, '127.0.0.1', yes)
})
const server = await createServer(
  mergeConfig(config, {
    configFile: false,
    envDir: scratch,
    server: {
      host: '127.0.0.1',
      port: 0,
      strictPort: false,
      open: false,
      proxy: { '/api': `http://127.0.0.1:${api.address().port}` },
    },
    plugins: [
      {
        name: 'controlled-pre-sdk-loading-failure',
        enforce: 'pre',
        load(id) {
          if (original && id.split('?')[0] === resolve('src/pose/workerInference.ts'))
            return original
        },
        configureServer(server) {
          server.middlewares.use((req, res, next) => {
            if (req.url?.split('?')[0] !== '/__bodyguard/inference-worker.js' || mode === 'normal')
              return next()
            res.setHeader('Content-Type', 'application/javascript')
            res.setHeader('Content-Security-Policy', SDK_DEV_POLICY)
            res.setHeader('Cache-Control', 'no-store')
            // Neither controlled script can initialize an SDK or acknowledge native cleanup.
            res.end(
              mode === 'script-error'
                ? 'throw new Error("Controlled worker loading failure")'
                : 'self.onmessage = () => {}',
            )
          })
        },
      },
    ],
  }),
)
try {
  await server.listen()
  const origin = server.resolvedUrls.local[0]
  browser = await chromium.launch({
    channel: 'chrome',
    args: ['--use-fake-device-for-media-stream', '--use-fake-ui-for-media-stream'],
    proxy: { server: 'http://127.0.0.1:1', bypass: `<-loopback>,${new URL(origin).host}` },
  })
  for (const fault of ['never-boots', 'script-error']) {
    const context = await browser.newContext({
      viewport: { width: 1440, height: 1100 },
      serviceWorkers: 'block',
    })
    context.on('request', (req) => {
      if (new URL(req.url()).origin !== new URL(origin).origin) external++
    })
    await context.addInitScript(() => {
      const camera = navigator.mediaDevices.getUserMedia.bind(navigator.mediaDevices),
        NativeWorker = Worker
      window.bootProbe = { streams: [], calls: 0, workers: 0, terminated: 0, events: [] }
      navigator.mediaDevices.getUserMedia = async (options) => {
        window.bootProbe.calls++
        const stream = await camera(options)
        window.bootProbe.streams.push(stream)
        return stream
      }
      const note = (direction, value) => {
        if (
          window.bootProbe.events.length < 50 &&
          ['boot', 'init', 'ready', 'frame', 'result', 'close', 'closed'].includes(value?.type)
        )
          window.bootProbe.events.push({
            direction,
            type: value.type,
            clean: value.type === 'closed' ? value.clean === true : null,
            at: performance.now(),
          })
      }
      window.Worker = class extends NativeWorker {
        constructor(url, options) {
          super(url, options)
          window.bootProbe.workers++
          this.addEventListener('message', ({ data }) => note('receive', data))
        }
        postMessage(value, transfer) {
          note('send', value)
          super.postMessage(value, transfer)
        }
        terminate() {
          window.bootProbe.terminated++
          super.terminate()
        }
      }
    })
    const page = await context.newPage()
    page.on('pageerror', (e) => errors.push(e.name))
    const probe = () =>
      page.evaluate(() => ({
        calls: window.bootProbe.calls,
        workers: window.bootProbe.workers,
        terminated: window.bootProbe.terminated,
        tracksEnded: window.bootProbe.streams.every((stream) =>
          stream.getTracks().every((track) => track.readyState === 'ended'),
        ),
        events: window.bootProbe.events,
        stored: localStorage.length + sessionStorage.length,
      }))
    try {
      mode = fault
      await page.goto(`${origin}dev/coaching?motion=neck-side`)
      const panel = page.getByRole('region', { name: '카메라 준비 확인' })
      await panel.getByRole('button', { name: '카메라 확인', exact: true }).click()
      await panel.getByRole('alert').waitFor({ timeout: 16000 })
      visibleError = await panel.getByRole('alert').innerText()
      traces.push({ fault, phase: 'failed', ...(await probe()), visibleError })
      await page.screenshot({ path: `${scratch}/${label}-${fault}-failed.png` })
      assert.equal(visibleError, '분석 연결을 준비하지 못했습니다. 다시 확인해 주세요.')
      assert.equal(
        await panel.getByRole('button', { name: '다시 확인', exact: true }).isEnabled(),
        true,
      )
      const failed = await probe()
      assert.equal(failed.tracksEnded, true)
      assert.equal(failed.workers, failed.terminated)
      assert.equal(failed.calls, 1)
      assert.equal(failed.events.filter((e) => e.direction === 'send').length, 0)
      assert.equal(failed.stored, 0)
      mode = 'normal'
      await panel.getByRole('button', { name: '다시 확인', exact: true }).click()
      await panel
        .getByRole('button', { name: '카메라 끄기', exact: true })
        .waitFor({ timeout: 16000 })
      assert.equal(await page.getByRole('button', { name: '코칭 준비 중' }).isDisabled(), true)
      await panel.getByRole('button', { name: '카메라 끄기', exact: true }).click()
      await panel.getByRole('button', { name: '카메라 확인', exact: true }).waitFor()
      const stopped = await probe()
      traces.push({ fault, phase: 'explicit-retry-stopped', ...stopped })
      assert.equal(stopped.calls, 2)
      assert.equal(stopped.workers, stopped.terminated)
      assert.equal(stopped.tracksEnded, true)
      assert.equal(stopped.stored, 0)
      assert(
        stopped.events.some((e) => e.direction === 'receive' && e.type === 'closed' && e.clean),
      )
      const flow = stopped.events.map((e) => `${e.direction}:${e.type}`)
      for (const event of [
        'receive:boot',
        'send:init',
        'receive:ready',
        'send:frame',
        'receive:result',
        'send:close',
        'receive:closed',
      ])
        assert(flow.includes(event))
      assert(flow.indexOf('receive:boot') < flow.indexOf('send:init'))
      assert.equal(page.workers().length, 0)
      await page.screenshot({ path: `${scratch}/${label}-${fault}-recovered.png` })
      cases.push(
        `${fault}: pre-SDK failure is recoverable; explicit retry uses actual heavy/face/hands SDK and clean native closure without reload`,
      )
    } finally {
      await context.close()
    }
  }
  assert.equal(external, 0)
  assert.equal(errors.length, 0)
} catch (e) {
  failure = { name: e.name, message: e.message.split('\n')[0] }
} finally {
  const version = browser?.version()
  await browser?.close()
  await server.close()
  await new Promise((yes) => api.close(yes))
  const after = snapshot()
  assert.deepEqual(before, after)
  writeFileSync(
    `${scratch}/${label}.json`,
    JSON.stringify(
      {
        method:
          'actual React/Chrome/native generated fake-device; controlled pre-SDK scripts and account API; not original intermittent P2 reproduction',
        variant: variant ?? 'working-source',
        originalParentSha256: original ? sha(original) : null,
        version,
        before,
        after,
        cases,
        traces,
        visibleError,
        external,
        errors,
        failure,
        actualPeople: false,
        actualPermissionPopup: false,
        originalIntermittentP2Resolved: false,
      },
      null,
      2,
    ),
    { flag: 'wx' },
  )
}
console.log(JSON.stringify({ output: `${scratch}/${label}.json`, passed: cases.length, failure }))
if (failure) process.exitCode = 1
